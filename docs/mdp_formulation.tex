\section{Notation}
\begin{align*}
&\text{Pursuer position } \bm{p}^P_t=[x^P_t,y^P_t]^\top,\; \text{velocity } \bm{v}^P_t,\; \text{acceleration } \bm{a}^P_t.\\
&\text{Evader position } \bm{p}^E_t,\; \text{velocity } \bm{v}^E_t,\; \text{heading } \psi^E_t.\\
&\text{Relative position } \bm{r}_t=\bm{p}^E_t-\bm{p}^P_t,\; d_t=\|\bm{r}_t\|,\; \hat{\bm{r}}_t=\bm{r}_t/(d_t+10^{-8}).\\
&\text{Relative velocity } \Delta\bm{v}_t=\bm{v}^E_t-\bm{v}^P_t,\; v^{\text{close}}_t=-\Delta\bm{v}_t^\top\hat{\bm{r}}_t.\\
&\text{World bound } [-W,W]^2,\; W=10.\quad \text{Catch radius } r_c=0.5.\\
&\text{Time step } \Delta t=0.05,\; T_{\max}=500.\quad \gamma=0.99.
\end{align*}

\section{Low-level MDP (Environment-level control)}
State space $\mathcal{S}_L\subset\mathbb{R}^{22}$ (clipped to $[-10,10]$):
\begin{align*}
\bm{s}^L_t = (&x^P_t/W,\; y^P_t/W,\; v^P_{x,t}/v_{\max},\; v^P_{y,t}/v_{\max},\; r_{x,t}/W,\; r_{y,t}/W,\\
&\Delta v_{x,t}/v_{\max},\; \Delta v_{y,t}/v_{\max},\; d_t/W,\; v^{\text{close}}_t/v_{\max},\; \hat{v}^E_{x,t},\; \hat{v}^E_{y,t},\\
&\text{dist}^{P}_{\text{top}}/W,\; \text{dist}^{P}_{\text{bottom}}/W,\; \text{dist}^{P}_{\text{left}}/W,\; \text{dist}^{P}_{\text{right}}/W,\\
&\text{dist}^{E}_{\text{boundary}}/W,\; \|\bm{v}^P_t\|/v_{\max},\; \|\bm{v}^E_t\|/v_{\max},\; \cos\angle(\bm{v}^P_t,\bm{r}_t),\; a^P_{x,t}/a_{\max},\; a^P_{y,t}/a_{\max}).
\end{align*}

Action space $\mathcal{A}_L=\{0,\dots,8\}$ (discrete thrust):
\begin{itemize}
  \item $0$: zero thrust; $1\dots 8$: full thrust in $\theta_k = (k-1)\tfrac{\pi}{4}$, thrust $\bm{\tau}=F_{\max}[\cos\theta_k,\sin\theta_k]$.
\end{itemize}

Transition $P_L$: pursuer uses 2D AUV dynamics
\begin{align*}
M \dot{\bm{v}} &= \bm{\tau} - D_l \bm{v} - D_q \|\bm{v}\|\bm{v},\\
\bm{a} &= \frac{\bm{\tau} + \bm{F}_{\text{damp}}}{M},\quad \|\bm{a}\|\le a_{\max},\; \|\bm{v}\|\le v_{\max},\\
\bm{v}_{t+1} &= \text{clip}(\bm{v}_t + \bm{a}\Delta t, v_{\max}),\\
\bm{p}_{t+1} &= \bm{p}_t + \tfrac{1}{2}(\bm{v}_t+\bm{v}_{t+1})\Delta t,
\end{align*}
with RK4 integration and boundary reflection (velocity scaled by $0.3$ when clamped).

Evader policy (stochastic exogenous) produces desired accel $\bm{a}^E_t$ via scripted modes (random/simple/smart/expert), then obeys turn-rate limit $|\Delta\psi|\le 0.1$ rad and same damped dynamics with $(v^E_{\max}, a^E_{\max})$ scaled by difficulty.

Reward $R_L(\bm{s}^L_t,a_t,\bm{s}^L_{t+1})=\sum_i r_i$:
\begin{align*}
&r_{\text{improve}} = w_1(d_{t-1}-d_t),\; w_1=5.0;\quad r_{\text{close}} = w_2\,\text{clip}(v^{\text{close}}_t,-1,1),\; w_2=1.0;\\
&r_{\text{align}} = w_3\, \cos\angle(\bm{v}^P_t,\bm{r}_t)\mathbb{1}_{\|\bm{v}^P_t\|>0.1},\; w_3=0.5;\\
&r_{\text{trap}} = w_4\,\tfrac{2-\text{dist}^E_{\text{boundary}}}{2}\, \mathbb{1}_{\text{dist}^E_{\text{boundary}}<2},\; w_4=0.3;\\
&r_{\text{time}} = -c_t,\; c_t=0.05;\quad r_{\text{wall}}=-c_b\,\mathbb{1}_{\text{hit boundary}},\; c_b=0.1;\\
&r_{\text{prox}} = 0.2(2-d_t)\,\mathbb{1}_{d_t<2};\quad r_{\text{catch}} = w_c\,\mathbb{1}_{d_t<r_c},\; w_c=100.
\end{align*}

Termination: $d_t<r_c$ (catch, terminal) or $t\ge T_{\max}$ (truncation). Discount $\gamma=0.99$.

\section{High-level MDP (Macro-action planner)}
State $\mathcal{S}_H\subset\mathbb{R}^6$:
\begin{align*}
\bm{s}^H_t = (\bm{r}_t/W,\; \bm{v}^P_t/2,\; \bm{v}^E_t/2).
\end{align*}
Action $\mathcal{A}_H=\{0,\dots,8\}$ mapped to waypoint $\bm{g}_t$:
\begin{align*}
\bm{g}_t = \begin{cases}
\bm{p}^E_t, & a_t=0,\\
\bm{p}^P_t + d_w[\cos\theta_k,\sin\theta_k], & a_t=k,\; k\in\{1..8\},\; d_w=2.0,\; \theta_k=(k-1)\tfrac{\pi}{4}.
\end{cases}
\end{align*}

High-level transition is semi-MDP: each macro-action invokes low-level controller for $K=5$ steps; cumulative reward $\tilde{r}_t=\sum_{j=0}^{K-1} r_{t+j}$ and termination if environment terminates within those steps. Effective transition kernel $P_H$ is induced by $P_L$ and the fixed low-level policy $\pi_L$ (pretrained, frozen).

\section{Option-style joint MDP}
Define option $o_t\in\mathcal{A}_H$ with duration $K$; low-level policy $\pi_L(\cdot\mid \bm{s}^L, o_t)$ is fixed. The induced SMDP tuple $(\mathcal{S}_H,\mathcal{A}_H,P_H,R_H,\gamma)$ has
\begin{align*}
P_H(\bm{s}^H_{t+1}\mid \bm{s}^H_t, o_t) &= \int P_L^{(K)}(\bm{s}^L_{t+K}\mid \bm{s}^L_t, o_t)\, d\bm{s}^L,\\
R_H(\bm{s}^H_t,o_t) &= \mathbb{E}\bigg[\sum_{j=0}^{K-1} \gamma^j r_{t+j} \mid \bm{s}^H_t,o_t \bigg].
\end{align*}
When optimization is separable (bottom frozen), the high-level objective maximizes
\begin{align*}
J_H = \mathbb{E}_{\pi_H,\pi_L}\Bigg[\sum_{t} \gamma^{tK} R_H(\bm{s}^H_t,o_t)\Bigg].
\end{align*}

\section{PPO Objectives}
High-level PPO (discrete):
\begin{align*}
L_{\text{actor}} &= -\mathbb{E}\big[\min(r_t\hat{A}_t,\; \text{clip}(r_t,1-\epsilon,1+\epsilon)\hat{A}_t)\big],\\
L_{\text{critic}} &= \tfrac{1}{2}\mathbb{E}[(V_\phi(\bm{s}^H_t)-\hat{R}_t)^2],\\
L &= L_{\text{actor}} + c_v L_{\text{critic}} - c_e H(\pi_\theta(\cdot\mid\bm{s}^H_t)),
\end{align*}
with GAE $\hat{A}_t$ and returns $\hat{R}_t=\hat{A}_t+V_\phi(\bm{s}^H_t)$. Low-level PPO shares the same form on $(\mathcal{S}_L,\mathcal{A}_L)$.

\section{Curriculum and Asymmetry}
Difficulty $\delta\in[0,1]$ rescales evader capability: $v^E_{\max,\text{eff}}=\max(v^E_{\max}\alpha, v^P_{\max}+0.05)$, $a^E_{\max,\text{eff}}=\min(a^E_{\max}\alpha, a^P_{\max}-0.05)$ with $\alpha=0.5+0.5\delta$. Initial distance sampled uniformly from $[d_{\min},d_{\max}]$ per preset (easy/medium/hard/expert). Evader policy switches among scripted modes; thus $P_L$ is Markov but stochastic via exogenous action generation.

\section{Summary of MDP Tuples}
\paragraph{Low-level} $\langle \mathcal{S}_L, \mathcal{A}_L, P_L, R_L, \gamma, T_{\max}\rangle$ with continuous state, 9 discrete actions, deterministic dynamics plus stochastic evader drive, rich shaped reward, termination on catch or horizon.

\paragraph{High-level} $\langle \mathcal{S}_H, \mathcal{A}_H, P_H, R_H, \gamma, K\rangle$ SMDP with macro-actions (waypoints), horizon in macro-steps $T_{\max}/K$, reward aggregated from low-level rollouts.
